<input type="text" id="itemInput" placeholder="Neues Produkt..."
onkeypress="if(event.key === 'Enter') addItem()">
<button onclick="addItem()">+</button>
</div>
<ul id="shoppingList"></ul>
<script>
document.addEventListener('DOMContentLoaded', loadList);
function getList() { return JSON.parse(localStorage.getItem('my_shopping_list') || '[]'); }
function saveList(list) { localStorage.setItem('my_shopping_list', JSON.stringify(list)); }
function loadList() {
const list = getList(); const ul = document.getElementById('shoppingList');
ul.innerHTML = '';
if (list.length === 0) { ul.innerHTML = '<li style="justify-content: center; color: #888;">Liste ist leer</li>'; return; }
list.forEach((item, index) => {
const li = document.createElement('li'); if (item.completed)
li.classList.add('completed');
li.innerHTML = `<span class="text-span" onclick="toggleItem(${index})">${item.text}</span><button class="delete-btn" onclick="removeItem(${index})">Löschen</button>`;
ul.appendChild(li);
});
}
function addItem() {
const input = document.getElementById('itemInput'); const text =
input.value.trim(); if (!text) return;
const list = getList(); list.push({ text: text, completed: false });
saveList(list); input.value = ''; loadList();
}
function toggleItem(index) { const list = getList(); list[index].completed = !list[index].completed; saveList(list); loadList(); }
function removeItem(index) { const list = getList(); list.splice(index, 1); saveList(list); loadList(); }
</script>
